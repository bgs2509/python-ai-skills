"""Regression suite for the branch-name-check PreToolUse hook.

The hook is the enforcement half of the N-BRANCH rule (claude-home/CLAUDE.md ->
"Naming"): it denies Bash git commands and EnterWorktree calls that create a branch
whose name has no `branch` segment, and must never block ordinary git work.

Hook contract: exit 2 = denied, exit 0 = allowed.

Run: pytest claude-home/hooks/test_branch_name_check.py
"""

import json
import os
import subprocess
from pathlib import Path

import pytest

HOOK = Path(__file__).resolve().parent / "branch-name-check.sh"
# No GIT_* variable may leak in: pre-commit runs this suite with GIT_INDEX_FILE (and
# possibly GIT_DIR) set, which would point `git init` and `git rev-parse` at the real repository.
ENV = {k: v for k, v in os.environ.items() if not k.startswith("GIT_")}


def make_repo(path):
    subprocess.run(["git", "init", "-q", str(path)], check=True, capture_output=True, env=ENV)
    return path


@pytest.fixture(scope="module")
def repo(tmp_path_factory):
    return make_repo(tmp_path_factory.mktemp("repo"))


@pytest.fixture
def home(tmp_path):
    """Fake HOME: the block journal (~/.claude/hook-stats/blocks.jsonl) never touches the real one."""
    path = tmp_path / "home"
    path.mkdir()
    return path


def run_hook(stdin, home):
    return subprocess.run(
        ["bash", str(HOOK)],
        input=stdin if isinstance(stdin, str) else json.dumps(stdin),
        capture_output=True,
        text=True,
        timeout=15,
        env={**ENV, "HOME": str(home)},
    )


def bash(command, cwd):
    return {"tool_name": "Bash", "tool_input": {"command": command}, "cwd": str(cwd)}


def journal(home):
    path = home / ".claude" / "hook-stats" / "blocks.jsonl"
    return [json.loads(line) for line in path.read_text().splitlines()] if path.exists() else []


BLOCKED = [
    # checkout / switch
    "git checkout -b feat/gigaam",
    "git checkout -B hotfix",
    "git checkout --orphan=docs-site",
    "git checkout --orphan docs-site",
    "git checkout -b x -- file",
    "git switch -c lazy-fox",
    "git switch --create=x",
    "git switch -cx",
    # prefixes, global options, chains
    "git -C /tmp/r switch -c x",
    "git add -A && git checkout -b x",
    "cd /tmp; git switch -c x",
    "VAR=1 git checkout -b x",
    "command git checkout -b x",
    "/usr/bin/git checkout -b x",
    "rtk proxy git switch -c x",
]

ALLOWED = [
    # names with the segment
    "git checkout -b feat/branch-x",
    "git switch -c fix/branch-y",
    # not creating
    "git checkout main",
    "git checkout -- file",
    "git checkout --track origin/lazy-fox",
    "git switch lazy-fox",
    "git push -u origin HEAD",
    "git status",
    # text, not commands
    'git commit -m "use git checkout -b x"',
    "echo 'git checkout -b x'",
    'grep -rn "checkout -b" docs/',
    # empty or unparsable
    "ls",
    "",
    "git checkout -b 'unbalanced",
    "git checkout -b",
]


@pytest.mark.parametrize("command", BLOCKED)
def test_branch_without_segment_is_denied(command, repo, home):
    result = run_hook(bash(command, repo), home)
    assert result.returncode == 2, f"hook allowed: {command!r}"
    assert "N-BRANCH" in result.stderr


@pytest.mark.parametrize("command", ALLOWED)
def test_ordinary_command_is_allowed(command, repo, home):
    result = run_hook(bash(command, repo), home)
    assert result.returncode == 0, f"hook blocked: {command!r}"
    assert result.stdout == "" and result.stderr == ""
    assert journal(home) == []


def test_deny_message_names_the_form_and_the_name(repo, home):
    result = run_hook(bash("git checkout -b lazy-fox", repo), home)
    assert "'lazy-fox'" in result.stderr
    assert "<type>/branch-<slug>" in result.stderr
    assert "feat, fix, docs, style, refactor, test, chore, perf, build, ci" in result.stderr
    assert "N-BRANCH: off" in result.stderr
    assert "machine-readable" not in result.stderr


def test_every_offending_name_is_listed(repo, home):
    result = run_hook(bash("git checkout -b a && git switch -c b", repo), home)
    assert "'a, b'" in result.stderr


def test_deny_is_journaled(repo, home):
    run_hook(bash("git checkout -b lazy-fox", repo), home)
    (record,) = journal(home)
    assert record["hook"] == "branch-name-check"
    assert record["kind"] == "bash-git"
    assert record["command"] == "git checkout -b lazy-fox"


@pytest.mark.parametrize("stdin", ["", "not json", "[]", "42", "{}", '{"tool_name": "Bash"}'])
def test_malformed_input_is_allowed(stdin, home):
    assert run_hook(stdin, home).returncode == 0


def test_other_tools_are_ignored(repo, home):
    payload = {"tool_name": "Read", "tool_input": {"command": "git checkout -b x", "name": "x"}, "cwd": str(repo)}
    assert run_hook(payload, home).returncode == 0

```python
"""PreToolUse hook: deny creating a git branch whose name has no `branch` segment.

Enforcement half of the N-BRANCH rule (claude-home/CLAUDE.md -> "Naming"; the
rule text lives only there, bd python-ai-skills-31d). Checked tools:

  - Bash: git commands that create a branch (checkout -b/-B/--orphan,
    switch -c/-C/--create/--force-create/--orphan, branch <name>,
    branch -m/-M/-c/-C <old> <new>, worktree add [-b/-B <name>] <path>).
  - EnterWorktree: tool_input.name (a call with only `path` enters an existing worktree).

Contract:
  - stdin: PreToolUse JSON (tool_name, tool_input, cwd).
  - exit 0: allow, silent. Any unparsable input, missing field or exception also
    ends in exit 0: only a verified non-compliant name is denied.
  - exit 2: deny, reason on stderr; one record is appended to
    ~/.claude/hook-stats/blocks.jsonl first.
  - Override: the line `N-BRANCH: off` under `## Naming overrides` in
    <git toplevel of cwd>/AGENTS.md; read only after an offending name is found.
"""

import datetime
import json
import os
import re
import shlex
import sys

# Rule constants. The wording of the rule is in claude-home/CLAUDE.md -> "Naming"
# (segment boundaries) and -> "Commit Convention" (types); keep these in step.
SEGMENT = re.compile(r"(^|[-/_+])branch($|[-/_+])")
FORM = "<type>/branch-<slug>"
TYPES = ("feat", "fix", "docs", "style", "refactor", "test", "chore", "perf", "build", "ci")

# Shell splitting, as in block-no-verify.sh (second consumer; extract on a third).
SEPARATORS = {"&&", "||", ";", "|", "&"}
GIT_OPTIONS_WITH_VALUE = {"-C", "-c", "--git-dir", "--work-tree", "--namespace", "--exec-path"}
PREFIX_WORDS = {"command", "env", "nice", "time"}
ASSIGNMENT = re.compile(r"^[A-Za-z_][A-Za-z0-9_]*=")

DENY = """[branch-name-check] Refusing to create branch '{names}': no `branch` segment.

Naming rule N-BRANCH (~/.claude/CLAUDE.md -> "Naming"): form {form},
<type> one of {types} (Commit Convention).
Worktrees: pass that form as the EnterWorktree name (Claude Code adds `worktree-` and turns `/` into `+`).
Project override: a line `N-BRANCH: off` under `## Naming overrides` in {toplevel}/AGENTS.md.
"""


def segments(tokens):
    """Split a token list into the commands of one shell line."""
    cur = []
    for tok in tokens:
        if tok in SEPARATORS:
            yield cur
            cur = []
        elif tok.endswith(";"):  # shlex keeps `;` glued to the previous word
            if tok.rstrip(";"):
                cur.append(tok.rstrip(";"))
            yield cur
            cur = []
        else:
            cur.append(tok)
    yield cur


def git_subcommand(segment):
    """(subcommand, its arguments) when the segment is a git command, else None."""
    i = 0
    while i < len(segment):
        if ASSIGNMENT.match(segment[i]) or segment[i] in PREFIX_WORDS:
            i += 1
        elif segment[i:i + 2] == ["rtk", "proxy"]:
            i += 2
        else:
            break
    if i >= len(segment) or not (segment[i] == "git" or segment[i].endswith("/git")):
        return None
    i += 1
    while i < len(segment):
        if segment[i] in GIT_OPTIONS_WITH_VALUE:
            i += 2
        elif segment[i].startswith("-"):
            i += 1
        else:
            return segment[i], segment[i + 1:]
    return None


def option_value(args, flags):
    """Value of the first option of `flags` (`-b x`, `-bx`, `--orphan x`, `--orphan=x`), else None."""
    for i, tok in enumerate(args):
        if tok == "--":
            return None
        for flag in flags:
            if tok == flag:
                return args[i + 1] if i + 1 < len(args) else ""
            if flag.startswith("--") and tok.startswith(flag + "="):
                return tok[len(flag) + 1:]
            if not flag.startswith("--") and not tok.startswith("--") and tok.startswith(flag):
                return tok[len(flag):]
    return None


def new_branch_name(sub, args):
    if sub == "checkout":
        return option_value(args, ("-b", "-B", "--orphan"))
    if sub == "switch":
        return option_value(args, ("-c", "-C", "--create", "--force-create", "--orphan"))
    return None


def offending_bash(command):
    """New branch names without the segment in a Bash command line."""
    if "git" not in command:
        return []
    try:
        tokens = shlex.split(command, posix=True)
    except ValueError:
        return []
    names = []
    for segment in segments(tokens):
        found = git_subcommand(segment)
        name = new_branch_name(*found) if found else None
        if name and not SEGMENT.search(name):
            names.append(name)
    return names


def log_block(kind, text):
    """Append one record to the block journal; a write error never changes the verdict."""
    try:
        directory = os.path.expanduser("~/.claude/hook-stats")
        os.makedirs(directory, exist_ok=True)
        record = {
            "ts": datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds"),
            "hook": "branch-name-check",
            "cwd": os.getcwd(),
            "kind": kind,
            "command": text[:300],
        }
        with open(os.path.join(directory, "blocks.jsonl"), "a", encoding="utf-8") as journal:
            journal.write(json.dumps(record, ensure_ascii=False) + "\n")
    except OSError:
        pass


def main():
    try:
        data = json.load(sys.stdin)
        if data["tool_name"] != "Bash":
            return 0
        text = data["tool_input"]["command"]
        names = offending_bash(text)
        if not names:
            return 0
        toplevel = data.get("cwd") or os.getcwd()
    except Exception:  # fail open on anything unexpected: only a verified name is denied
        return 0
    log_block("bash-git", text)
    message = DENY.format(names=", ".join(names), form=FORM, types=", ".join(TYPES), toplevel=toplevel)
    sys.stderr.write(message)
    return 2


if __name__ == "__main__":
    sys.exit(main())
```

def newlines_to_separators(command):
    """Replace each newline outside quotes by ` ; ` (shlex treats a newline as a space)."""
    out, quote, i = [], "", 0
    while i < len(command):
        ch = command[i]
        if ch == "\\" and quote != "'":  # an escaped character, a newline included, is never a separator
            out.append(command[i:i + 2])
            i += 2
            continue
        if quote:
            if ch == quote:
                quote = ""
        elif ch in "'\"":
            quote = ch
        elif ch == "\n":
            out.append(" ; ")
            i += 1
            continue
        out.append(ch)
        i += 1
    return "".join(out)

## Deviations from the design

1. **D1, newline handling (Task 6).** Design 2.3 step 2 calls `shlex.split` on the raw command; `shlex` treats a newline as a space, so a creation on the second line of a multi-line command is invisible. Added `newlines_to_separators` (about 20 lines) and its tests. If the orchestrator rejects it, drop Task 6 and the three "several lines" test entries; nothing else depends on it.
2. **D2, the git token position (Task 2).** Design 2.3 step 5 strips prefix words and then says "find the first token equal to `git` or ending in `/git`". A free search after the strip makes the strip dead code and would deny `echo git checkout -b x` and `man git branch foo`. Implemented: after the prefix strip the first token must be `git` (or end in `/git`). Not covered (fail-open, as risk 4): `sudo git ...`, `timeout 5 git ...`, `xargs git ...`.
3. **D3, the override note assertion (Task 5).** Design 2.7 says stderr contains `not machine-readable`; the stderr template in design 2.2 reads "...but only `N-BRANCH: off` is machine-readable." The test asserts the template text; the template is unchanged.
4. **D4, the test repository (Task 2).** Design 2.7 says `git init` plus one commit; `git rev-parse --show-toplevel` needs no commit, so the fixture only runs `git init -q` (no user config needed). The test environment strips every `GIT_*` variable because pre-commit runs this suite with `GIT_INDEX_FILE` set.
5. **D5, commit slicing.** One commit per behaviour slice (Tasks 2-6) instead of a single feature commit, so each commit is green on its own; the launcher is committed at mode 644 and fixed in Task 7.
6. **D6, `kind` text for an empty `EnterWorktree` call.** The journal `command` field holds `<generated>` when the call has neither `name` nor `path` (the design leaves the value open).
